\section{Introdução}
\begin{frame}{Introdução}{Contexto e Motivação}
  \begin{itemize}
    \item Importância da qualidade de software em sistemas críticos.
    \item Limitações dos testes tradicionais para explorar todos os cenários possíveis.
    \item Motivação para o uso de Model Checking como técnica complementar.
    \item Contexto do \emph{e-Título} como sistema crítico do TSE.
  \end{itemize}
\end{frame}

\begin{frame}{Introdução}{Definição do Problema}
  \begin{itemize}
    \item Dificuldade em identificar falhas emergentes de interações entre componentes.
    \item Ausência de integração consolidada entre testes automatizados e verificação formal.
    \item Pergunta de pesquisa: a integração de Model Checking com testes melhora a detecção de falhas?
  \end{itemize}
\end{frame}

\section{Justificativa e Objetivos}
\begin{frame}{Justificativa}{Relevância}
  \begin{itemize}
    \item \emph{e-Título} atende milhões de eleitores e exige alta confiabilidade.
    \item Fluxos complexos (síncronos e assíncronos) aumentam riscos não cobertos por testes convencionais.
    \item Model Checking complementa testes ao identificar interações e estados indesejados.
  \end{itemize}
\end{frame}

\begin{frame}{Objetivos}{Geral e Específicos}
  \begin{itemize}
    \item \textbf{Objetivo Geral:} Guiar a definição de testes automatizados a partir de análises por Model Checking.
    \item \textbf{Objetivos Específicos:}
      \begin{itemize}
        \item Investigar limitações dos testes automatizados atuais.
        \item Explorar a viabilidade do Model Checking como complemento.
        \item Definir um método semi-automatizado para geração de testes.
        \item Aplicar e validar a metodologia no estudo de caso do \emph{e-Título}.
      \end{itemize}
  \end{itemize}
\end{frame}

\section{Estado da Arte}
\begin{frame}{Estado da Arte}{Visão Geral}
  \begin{itemize}
    \item Revisão baseada em survey sobre testes de APIs REST e trabalhos posteriores.
    \item Crescimento recente do uso de IA (LLMs) para geração e oráculos de teste.
    \item Lacuna: poucas integrações entre técnicas formais (model checking) e frameworks de teste.
  \end{itemize}
\end{frame}

\begin{frame}{Estado da Arte}{Taxonomia para APIs REST}
  \begin{itemize}
    \item \textbf{Cobertura:} esquema (OpenAPI), código, métricas especializadas.
    \item \textbf{Detecção de falhas:} erros 5xx, violações de esquema, violação de regras definidas.
    \item \textbf{Desempenho:} tempo de resposta e latência (menos explorado).
  \end{itemize}
\end{frame}

\begin{frame}{Estado da Arte}{Limitações Observadas}
  \begin{itemize}
    \item Testes tendem a focar em cenários previsíveis e "happy path".
    \item Oráculos fracos e restrições ocultas reduzem efetividade.
    \item Pouca exploração de verificação formal aplicada a APIs REST em contexto industrial.
  \end{itemize}
\end{frame}

\section{Fundamentação Teórica}
\begin{frame}{Fundamentação Teórica}{Testes de Software}
  \begin{itemize}
    \item Testes como atividades de Verificação, Validação e Teste (VV\&T).
    \item Diferença entre verificação (conforme especificação) e validação (produto certo).
    \item Limitações de testes unitários e importância de testes de integração para sistemas distribuídos.
  \end{itemize}
\end{frame}

\begin{frame}{Fundamentação Teórica}{Model Checking}
  \begin{itemize}
    \item Exploração exaustiva do espaço de estados de um modelo formal.
    \item Ferramentas: SPIN (Promela) e PRISM (modelos probabilísticos).
    \item Dependência da qualidade do modelo e da formulação de propriedades (LTL).
  \end{itemize}
\end{frame}

\begin{frame}{Fundamentação Teórica}{Integração com Testes}
  \begin{itemize}
    \item Model Checking identifica cenários logicamente possíveis que violam propriedades.
    \item Transformação de violações em contratos (Design by Contract) e casos de teste.
    \item Geração de testes que validam se o sistema implementado realmente evita esses cenários.
  \end{itemize}
\end{frame}

\section{Metodologia}
\begin{frame}{Metodologia}{Visão Geral}
  \begin{itemize}
    \item Abordagem iterativa (PDCA): modelagem, verificação, extração de caminhos, teste.
    \item Critérios de avaliação: cobertura, detecção de falhas e aplicabilidade prática.
    \item Ferramentas e recursos: SPIN, Promela, ferramentas de teste para APIs REST.
  \end{itemize}
\end{frame}

\begin{frame}{Metodologia}{Etapas}
  \begin{enumerate}
    \item Construção do modelo Promela do fluxo alvo (onboarding).
    \item Definição de propriedades LTL representando requisitos e invariantes.
    \item Execução do SPIN e análise de contra-exemplos / violações.
    \item Tradução de contra-exemplos em casos de teste automatizados (ex.: BDD).
  \end{enumerate}
\end{frame}

\section{Modelagem do Onboarding}
\begin{frame}{Modelagem}{Fluxos Modelados}
  \begin{itemize}
    \item Etapas modeladas: preenchimento de dados biográficos, carrossel de perguntas, senha e verificação biométrica.
    \item Uso de processos concorrentes em Promela para representar interações reais.
    \item Geração de grafo de estados para análise de propriedades e métricas.
  \end{itemize}
\end{frame}

\begin{frame}{Modelagem}{Métricas e Mapeamentos}
  \begin{itemize}
    \item Aplicação de métricas especializadas para avaliar representatividade do modelo.
    \item Mapeamento entre endpoints REST e transições do modelo (tabela de mapeamento).
    \item Importância do alinhamento entre implementação e modelo.
  \end{itemize}
\end{frame}

\section{Aplicação do SPIN e Propriedades}
\begin{frame}{SPIN e LTL}{Definição de Propriedades}
  \begin{itemize}
    \item Propriedades LTL formuladas para capturar comportamentos esperados (ex.: sequências inválidas).
    \item Verificação busca contra-exemplos, ciclos de aceitação e violações.
    \item Tradução das propriedades violadas em cláusulas de Design by Contract (DbC).
  \end{itemize}
\end{frame}

\begin{frame}{SPIN e LTL}{Exemplos e Observações}
  \begin{itemize}
    \item Observação de acceptance cycles e análise das causas.
    \item Extração de caminhos de execução a partir de contra-exemplos.
    \item Conversão dos passos do contra-exemplo em passos de um caso de teste BDD.
  \end{itemize}
\end{frame}

\section{Geração de Casos de Teste}
\begin{frame}{Geração de Testes}{Fluxo de Geração}
  \begin{itemize}
    \item Cada violação no modelo descreve um cenário lógico possível.
    \item Cenários são convertidos em scripts de teste automáticos ou em BDD (Given-When-Then).
    \item Integração com frameworks de teste para executabilidade contra a API real.
  \end{itemize}
\end{frame}

\begin{frame}{Geração de Testes}{Exemplo (Resumo)}
  \begin{itemize}
    \item Contra-exemplo: usuário que falha no carrossel e consegue pular etapa X.
    \item Conversão: sequência de chamadas REST e dados de entrada para reproduzir o cenário.
    \item Resultado: teste detecta comportamento indevido no sistema real.
  \end{itemize}
\end{frame}

\section{Resultados e Análises}
\begin{frame}{Resultados}{Principais Achados}
  \begin{itemize}
    \item Identificação de inconsistências no fluxo de onboarding do \emph{e-Título}.
    \item Tradução sistemática de LTL violadas para contratos DbC e casos de teste.
    \item Demonstração de aplicabilidade da metodologia em um caso real.
  \end{itemize}
\end{frame}

\begin{frame}{Resultados}{Limitações}
  \begin{itemize}
    \item Dependência da qualidade e fidelidade do modelo Promela.
    \item Escalabilidade do Model Checking para modelos muito grandes.
    \item Necessidade de revisão humana nos artefatos gerados (ex.: scripts de teste).
  \end{itemize}
\end{frame}

\section{Conclusão e Trabalhos Futuros}
\begin{frame}{Conclusão}{Síntese}
  \begin{itemize}
    \item A integração entre Model Checking e testes automatizados aumenta a robustez da validação.
    \item Método aplicável ao \emph{e-Título} com resultados promissores.
    \item Recomenda-se aumento da automação e ampliação do escopo (mais fluxos e serviços).
  \end{itemize}
\end{frame}

\begin{frame}{Trabalhos Futuros}{Direções}
  \begin{itemize}
    \item Automação completa da transformação de contra-exemplos em testes executáveis.
    \item Explorar modelos probabilísticos (PRISM) para situações estocásticas.
    \item Integração com ferramentas de IA para auxiliar na geração de dados de teste e oráculos.
  \end{itemize}
\end{frame}
