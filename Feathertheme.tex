\section{Introdução}
\begin{frame}{Motivação}{Contexto}
  \begin{itemize}
    \item APIs REST são base de sistemas modernos e críticos.
    \item Erros de integração frequentemente não são detectados por testes tradicionais.
    \item Testes manuais e exploratórios possuem baixa sistematização.
    \item Lacuna entre modelos formais e práticas de teste de software.
  \end{itemize}
\end{frame}

\begin{frame}{Introdução}{Estudo de Caso: e-Título}
  \begin{itemize}
    \item Aplicativo oficial da Justiça Eleitoral brasileira, com mais de 57 milhões de eleitores ativos.
    \item Arquitetura baseada em APIs REST, suportando processos críticos de identificação e emissão de documentos digitais.
    \item Alta complexidade nos fluxos de autenticação, combinando componentes síncronos e assíncronos.
    \item Demanda rigorosos requisitos de segurança, confiabilidade e disponibilidade.
  \end{itemize}
\end{frame}

\begin{frame}{Introdução}{Definição do Problema}
  \begin{itemize}
    \item Testes tradicionais operam apenas na implementação e não exploram exaustivamente interações emergentes.
    \item A eficácia do \textit{Model Checking} esbarra no distanciamento entre o modelo formal e a execução prática no ambiente de testes.
    \item \textbf{Problema central:} Faltam mecanismos que transformem cenários críticos descobertos por verificação formal em restrições diretamente executáveis no código das APIs.
  \end{itemize}
\end{frame}

\begin{frame}{Introdução}{Objetivos do Trabalho}
  \begin{itemize}
    \item \textbf{Objetivo Geral:} Desenvolver, avaliar e validar um método que guie a definição de contratos DbC a partir da análise de cenários via \textit{Model Checking} em sistemas críticos.
    \item \textbf{Objetivos Específicos:}
    \begin{enumerate}
        \item Investigar as limitações de técnicas automatizadas de teste.
        \item Explorar a viabilidade do \textit{Model Checking} como técnica complementar.
        \item Definir método para geração de contratos DbC baseados na exploração sistemática de estados.
        \item Aplicar a metodologia no estudo de caso do e-Título.
    \end{enumerate}
  \end{itemize}
\end{frame}

\section{Estado da Arte e Fundamentação}
\begin{frame}{Estado da Arte}{Revisão Bibliográfica}
  \begin{itemize}
    \item O cenário atual foca vastamente em estratégias de caixa-preta e, mais recentemente, IA/LLMs para automação de testes REST.
    \item Abordagens que empregam técnicas formais focam em invariantes e propriedades comportamentais, mas ainda são limitadas.
    \item \textbf{Oportunidade:} Beneficiar-se do rigor matemático na identificação de violações sutis, integrando os achados às práticas de desenvolvimento.
  \end{itemize}
\end{frame}

\begin{frame}{Fundamentação Teórica}{Model Checking e Design by Contract (DbC)}
  \begin{itemize}
    \item \textbf{Model Checking (SPIN/Promela):} Exploração sistemática e exaustiva de todos os estados do sistema em busca de falhas lógicas.
    \item \textbf{LTL (Linear Temporal Logic):} Linguagem de especificação de propriedades desejadas, verificando o tempo de execução e a progressão dos estados.
    \item \textbf{Design by Contract (DbC):} Formalização de acordos entre cliente e fornecedor (Módulos/APIs) usando pré-condições, pós-condições e invariantes.
  \end{itemize}
\end{frame}

\section{Metodologia}
\begin{frame}{Metodologia}{Abordagem Geral e PDCA}
  \begin{itemize}
    \item Pesquisa aplicada, descritiva e experimental baseada no sistema e-Título.
    \item Método iterativo estruturado no ciclo PDCA (\textit{Plan-Do-Check-Act}) para refinamento contínuo dos modelos.
    \item \textbf{O ciclo de verificação inclui:}
    \begin{enumerate}
        \item Modelagem comportamental em linguagem de alto nível (Promela).
        \item Verificação de propriedades via SPIN.
        \item Análise de violações (\textit{contraexemplos}) e ajustes.
        \item Extração das cláusulas DbC a partir da estrutura do modelo gerado.
    \end{enumerate}
  \end{itemize}
\end{frame}

\begin{frame}{Resultados Preliminares}{Critérios de Avaliação do Método}
  \begin{itemize}
    \item \textbf{Adequação da Modelagem:} Representatividade do modelo Promela comparado ao sistema em produção.
    \item \textbf{Expressividade Contratual:} Avaliação se a cláusula DbC cobre integralmente as restrições inferidas pelo modelo.
    \item \textbf{Taxa de Detecção:} Medir propriedades violadas encontradas pelo modelo que fugiram à suíte atual de testes.
    \item \textbf{Utilidade Prática:} Avaliação qualitativa da viabilidade do método junto aos desenvolvedores no TSE.
  \end{itemize}

\section{Conclusão e Próximos Passos}
\begin{frame}{Conclusão Parcial e Próximos Passos}
  \begin{itemize}
    \item O uso guiado de contraexemplos gerados pelo SPIN atua como fonte rica para a engenharia de requisitos comportamentais, definindo contratos DbC mais robustos.
    \item \textbf{Próximas Etapas (Cronograma):}
    \begin{itemize}
        \item Ampliar o escopo dos fluxos de \textit{onboarding} do e-Título.
        \item Realizar avaliação empírica das falhas detectadas usando os contratos gerados vs. testes tradicionais.
        \item Avançar na automatização parcial da tradução de Promela/LTL para especificações DbC.
        \item Submissão e defesa final da dissertação.
    \end{itemize}
  \end{itemize}
\end{frame}
